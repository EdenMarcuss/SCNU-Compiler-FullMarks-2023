int fact(int x) {
    if (x <= 1) return 1;
    return x * fact(x - 1);
}

float sum(float a[], int n) {
    float s;
    int i;
    s = 0.0;
    i = 0;
    do {
        s = s + a[i];
        i = i + 1;
    } while (i < n);
    return s;
}
